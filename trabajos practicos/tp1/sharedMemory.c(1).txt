#include <stdio.h>
#include <sys/shm.h>
#include <sys/stat.h>
#include <unistd.h>

#define TAM 1024

int main()
{
    int segment_id;      // un id del Segmento de Memoria compartida
    char *shared_memory; // un puntero a la Memoria Compartida, string
    // obtener un ID del segmento de memoria
    segment_id = shmget(1981, TAM, 0664 | IPC_CREAT);

    if (segment_id == -1)
    { // si surge algun problema
        fprintf(stderr, "Error al reservar la memoria");
    }
    // obtener un puntero al string de memoria Compartida
    shared_memory = (char *)shmat(segment_id, NULL, 0);

    printf("escribiendo en la memoria compartida\n");
    int i = 0;
    //for (i = 0; i < 10; i++)
    {
        sprintf(shared_memory, "HOLA! %d", i); // Escribir en el Shared Memory
        sleep(1);
    }
    
    // Enviar a la Pantalla lo que hay en la memoria compartida
    printf("%s\n", shared_memory);
    shmdt(shared_memory);               // Desconectar Shared Memory
    shmctl(segment_id, IPC_RMID, NULL); // Eliminar el espacio de direcciones 
    return 0;
}
